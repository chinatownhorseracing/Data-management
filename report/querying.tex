\section{Querying}


\subsection*{Ex 13}
\begin{lstlisting}
SELECT
    students.student_id,
    students.student_firstname,
    students.student_lastname,
    AVG(exam_results.exam_mark) AS avg_exam_mark
FROM
    students
    JOIN exam_results ON students.student_id = exam_results.student_id
    JOIN modules ON exam_results.module_id = modules.module_id
WHERE
    modules.module_id LIKE 'COMP%'
    AND students.year = 1
GROUP BY
    students.student_id
ORDER BY
    avg_exam_mark DESC
LIMIT
    5;
\end{lstlisting}

\subsection*{Ex 14}
\begin{lstlisting}
SELECT
    module_id,
    leader_firstname,
    leader_surname,
    faculty,
    average_mark
FROM
    (
        SELECT
            modules.module_id,
            lecturers.lecturer_firstname AS leader_firstname,
            lecturers.lecturer_surname AS leader_surname,
            lecturers.faculty,
            AVG(coursework_marks.mark) + AVG(exam_results.exam_mark) AS average_mark,
            ROW_NUMBER() OVER (
                PARTITION BY
                    lecturers.faculty
                ORDER BY
                    (
                        AVG(coursework_marks.mark) + AVG(exam_results.exam_mark)
                    ) DESC
            ) AS mark_rank_for_this_faculty
        FROM
            coursework_marks
            JOIN exam_results ON coursework_marks.module_id = exam_results.module_id
            JOIN modules ON exam_results.module_id = modules.module_id
            JOIN lecturers ON modules.leader = lecturers.lecturer_email
        GROUP BY
            exam_results.module_id
    )
WHERE
    mark_rank_for_this_faculty = 1;
\end{lstlisting}

\subsection*{Ex 15}
\begin{lstlisting}
SELECT
    module_lecturers.module_id,
    room_lecturers.building,
    room_lecturers.room
FROM
    exam_results
    JOIN module_lecturers ON exam_results.module_id = module_lecturers.module_id
    JOIN room_lecturers ON module_lecturers.lecturer_email = room_lecturers.lecturer_email
    JOIN rooms ON room_lecturers.building = rooms.building
    AND room_lecturers.room = rooms.room
GROUP BY
    module_lecturers.module_id,
    room_lecturers.building,
    room_lecturers.room
HAVING
    COUNT(DISTINCT exam_results.student_id) > rooms.capacity;
\end{lstlisting}
